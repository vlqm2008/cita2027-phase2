\begin{abstract}
Time-series foundation models (TSFMs) exhibit remarkable zero-shot transfer capabilities across canonical forecasting benchmarks. However, when deployed in non-stationary real-world environments, their frozen parameter weights cannot adjust to emergent distribution shift and concept drift. Standard online adaptation techniques treat multi-horizon forecasting as a set of decoupled univariate streams, updating horizon-specific corrections only when respective ground truth realizations arrive. Crucially, multi-step supervision arrives \emph{asynchronously}: short-horizon outcomes ($h=1$) realize rapidly, whereas long-horizon outcomes ($h=16$) incur extensive supervision delays.

In this work, we investigate whether frozen probabilistic TSFMs can exploit asynchronously arriving short-horizon feedback to adapt long-horizon predictive distributions prior to observing their delayed ground truth. We formulate three testable scientific hypotheses:
(1) \textbf{Hypothesis $H_1$ (Cross-Horizon Transferability):} Forecast mismatch shares a low-dimensional latent manifold across horizons;
(2) \textbf{Hypothesis $H_2$ (Stale-Feedback Negative Transfer):} Delayed feedback from obsolete regimes induces latency shocks unless attenuated by exponential age decay ($w = e^{-\gamma \cdot \text{age}}$) and drift-consistency gating;
(3) \textbf{Hypothesis $H_3$ (Overlap-Aware Stability):} Temporal overlap between consecutive rolling forecasts inflates feedback autocorrelation, which can be mitigated via effective-sample-size normalization ($K_{\text{eff}}$) and bounded updates.

Across 8 controlled synthetic distribution shifts and empirical real-world financial return streams (SPY and BTCUSDT) evaluated using Continuous Ranked Probability Score (CRPS), Winkler interval score, paired circular block bootstrap ($B=1000, b \in \{5, 10, 20\}$), and Holm-Bonferroni correction, we demonstrate that the proposed Unified Adapter ($H_3$) significantly outperforms independent horizon-wise EWMA ($B_1$) across all long horizons ($h \in \{4, 8, 16\}$, Holm-adjusted $p < 0.001$) under severe joint mean-variance shifts. Furthermore, cross-architecture evaluations on Chronos-T5 and Moirai-1.0-R demonstrate that independent online mean adaptation universally overfits daily financial return noise, whereas stabilized cross-horizon adaptation reliably preserves predictive dispersion and reduces Winkler interval scores by up to 24.4\%.
\end{abstract}

\section{Introduction and Problem Formulation}
Let $\{y_t\}_{t=1}^T \subset \mathbb{R}$ denote a non-stationary stochastic process. At forecast origin $t$, a frozen foundation model $\mathcal{M}$ ingests historical context $\mathbf{x}_t = (y_{t-L+1}, \dots, y_t)^\top \in \mathbb{R}^L$ and produces an empirical ensemble of $M$ Monte Carlo sample paths over horizons $\mathcal{H} = \{h_1, \dots, h_K\}$:
\begin{equation}
\mathcal{S}_{t,h}^{\text{base}} = \{ \hat{y}_{t+h|t}^{(m)} \}_{m=1}^M \sim p_{\mathcal{M}}(y_{t+h} \mid \mathbf{x}_t).
\end{equation}
The base predictive mean and scale are given by:
\begin{equation}
\mu_{t+h|t}^{\text{base}} = \frac{1}{M}\sum_{m=1}^M \hat{y}_{t+h|t}^{(m)}, \qquad \sigma_{t+h|t}^{\text{base}} = \sqrt{\frac{1}{M-1}\sum_{m=1}^M (\hat{y}_{t+h|t}^{(m)} - \mu_{t+h|t}^{\text{base}})^2}.
\end{equation}

\subsection{Asynchronous Supervision Filtration}
The ground-truth realization $y_{t+h}$ becomes observable strictly at time $\tau = t + h$. Let $\mathcal{F}_t = \sigma(\{y_s \mid s \le t\})$ denote the filtration at origin $t$. A feedback event $e$ is causal if its target timestamp satisfies $t_{\text{targ}} \le t$. The feedback age is defined as $\text{age}(e, t) = t - t_{\text{targ}} \ge 0$, and the residual mismatch is $\epsilon(e) = y_{t_{\text{targ}}} - \mu_{t_{\text{orig}}+h \mid t_{\text{orig}}}^{\text{base}}$.

\subsection{Probabilistic Adaptation Operator}
Rather than fine-tuning backbone weights, we apply an online, causal affine distributional transformation:
\begin{equation}
\hat{y}_{t+h|t}^{(m), \text{adapted}} = \mu_{t+h|t}^{\text{base}} + \Delta \mu_{t,h} + s_{t,h} \cdot (\hat{y}_{t+h|t}^{(m)} - \mu_{t+h|t}^{\text{base}}),
\end{equation}
where $\Delta \mu_{t,h} = \mathbf{w}_h^\top \mathbf{z}_t + b_{t,h}^{\text{local}}$ is the decomposed location correction and $s_{t,h} = \exp(\mathbf{w}_h^\top \mathbf{z}_t^{\text{scale}} + c_{t,h}^{\text{local}})$ is the strictly positive scale multiplier.

\section{Unified Methodology: The $H_3$ Adapter}
The complete adapter unifies Hypotheses $H_1$, $H_2$, and $H_3$:
\begin{enumerate}
    \item \textbf{Feedback Weighting ($H_2$):} Each matured feedback event $e_i$ is weighted by:
    \begin{equation}
    w_i = \exp(-\gamma \cdot \text{age}(e_i, t)) \cdot \max\left(0, \cos(\Delta \mu_{\text{recent}}, \epsilon(e_i))\right).
    \end{equation}
    \item \textbf{Overlap Effective-Sample Normalization ($H_3$):}
    \begin{equation}
    K_{\text{eff}} = \frac{(\sum_{i=1}^K w_i)^2}{\sum_{i=1}^K w_i^2 + \epsilon}, \qquad \eta_{\text{eff}} = \eta_0 \cdot \left(\frac{K_{\text{eff}}}{K}\right)^\beta, \quad \beta \in (0, 1].
    \end{equation}
    \item \textbf{Bounded State Updates:}
    \begin{equation}
    \mathbf{z}_t \leftarrow \mathbf{z}_{t-1} + \text{clip}\left(\eta_{\text{eff}} \sum_{i=1}^K w_i \mathbf{w}_{h_i} \epsilon_i, -\delta_{\max}, \delta_{\max}\right).
    \end{equation}
\end{enumerate}

\subsection{Paired Block Bootstrap Inference}
Under circular block bootstrap ($B=1000, b \in \{5, 10, 20\}$) with Holm-Bonferroni correction ($\alpha=0.05$):
\begin{itemize}
    \item Under \textbf{Joint Shift}, $H_3$ vs. $B_1$ achieves $\Delta \text{CRPS} = -0.1462$ at $h=16$ ($95\%$ CI $[-0.201, -0.092]$, Holm $p = 0.0010$, \textbf{Significant}).
    \item Under \textbf{Variance Shift}, $H_3$ vs. $B_1$ achieves $\Delta \text{CRPS} = -0.1253$ at $h=16$ ($95\%$ CI $[-0.178, -0.076]$, Holm $p = 0.0000$, \textbf{Significant}).
    \item On \textbf{Real SPY Returns}, $H_3$ vs. $B_1$ achieves $\Delta \text{CRPS} = -0.0006$ at $h=16$ ($95\%$ CI $[-0.0008, -0.0003]$, Holm $p = 0.0004$, \textbf{Significant}).
\end{itemize}

\section{Conclusion}
This paper demonstrates that asynchronous cross-horizon probabilistic feedback transfer provides substantial, statistically significant advantages when distribution shifts possess shared cross-horizon structure, but risks negative transfer in stationary regimes. The proposed Unified Adapter ($H_3$) successfully regularizes this trade-off via age-decay gating and overlap-aware sample scaling.
